import * as THREE from 'three'; import { FBXLoader } from 'three/examples/jsm/loaders/FBXLoader.js'; async function main() { const loader = new FBXLoader(); // Load base model const model = await loader.loadAsync('/assets/Hero_base.fbx'); console.log('=== HERO_BASE ==='); console.log('Type:', model.type); console.log('Children:', model.children.length); console.log('Has skeleton:', model.children.some(c => c.type === 'Bone' || (c as any).isBone)); // Find bones/skeleton const bones: THREE.Object3D[] = []; model.traverse((child) => { if ((child as any).isBone) bones.push(child); }); console.log('Bones found:', bones.length); if (bones.length > 0) { console.log('First 5 bone names:', bones.slice(0, 5).map(b => b.name)); console.log('Bone hierarchy sample:'); const rootBones = bones.filter(b => !b.parent || !(b.parent as any).isBone); rootBones.forEach(b => { console.log(` Root: ${b.name}, children: ${b.children.filter(c => (c as any).isBone).length}`); }); } // Find meshes const meshes: THREE.Mesh[] = []; model.traverse((child) => { if ((child as any).isMesh) meshes.push(child as THREE.Mesh); }); console.log('Meshes:', meshes.length); console.log('Animations:', model.animations?.length); if (model.animations?.length) { console.log('Anim names:', model.animations.map((a: any) => a.name)); console.log('Anim 0 tracks:', model.animations[0].tracks.length); console.log('Anim 0 first 3 tracks:', model.animations[0].tracks.slice(0, 3).map((t: any) => t.name)); } // Load each animation file for (const name of ['Idle', 'Run', 'Attack', 'Hit', 'Die']) { const anim = await loader.loadAsync(`/assets/${name}.fbx`); console.log(`\n=== ${name} ===`); console.log('Animations:', anim.animations?.length); if (anim.animations?.length) { console.log('Clip name:', anim.animations[0].name); console.log('Tracks:', anim.animations[0].tracks.length); console.log('Duration:', anim.animations[0].duration); if (anim.animations[0].tracks.length > 0) { console.log('First 3 tracks:', anim.animations[0].tracks.slice(0, 3).map((t: any) => t.name)); } } const animBones: THREE.Object3D[] = []; anim.traverse((child) => { if ((child as any).isBone) animBones.push(child); }); console.log('Bones in anim file:', animBones.length); if (animBones.length > 0) { console.log('First 5 bone names:', animBones.slice(0, 5).map(b => b.name)); } } } main().catch(err => { console.error('TEST ERROR:', err.message); console.error(err.stack); });
